import { compileScript, compileTemplate, compileStyleAsync, parse } from '@vue/compiler-sfc';
import * as Babel from '@babel/standalone';

function runInSandbox(code: string, sandbox: Record<string, any>) {
  const sandboxProxy = new Proxy(sandbox, {
    has: () => true, // permette l’uso di qualsiasi variabile definita nel sandbox
  });

  const func = new Function("sandbox", `
    with (sandbox) {
      "use strict";
      return (function() {
        ${code}
      })();
    }
  `);

  return func(sandboxProxy);
}

function rewriteImportsToModuleCache(
  code: string,
  moduleCacheVar: string = "moduleCache"
): string {
  function moduleVar(moduleName: string): string {
    // se il modulo è già una variabile in scope (perché preso da moduleCache), usalo direttamente
    return moduleName in (globalThis as any)[moduleCacheVar]
      ? moduleName
      : `${moduleCacheVar}["${moduleName}"]`;
  }

  // import { a, b as c } from "vue"
  code = code.replace(
    /import\s+\{([^}]+)\}\s+from\s+['"]([^'"]+)['"];?/g,
    (_, imports, moduleName) => {
      const names = imports
        .split(",")
        .map(s => s.trim())
        .filter(Boolean)
        .map(s => {
          if (s.includes(" as ")) {
            const [orig, alias] = s.split(" as ").map(x => x.trim());
            return `${orig}: ${alias}`;
          }
          return s;
        })
        .join(", ");
      return `const { ${names} } = ${moduleVar(moduleName)};`;
    }
  );

  // import defaultExport from "vue"
  code = code.replace(
    /import\s+([a-zA-Z0-9_$]+)\s+from\s+['"]([^'"]+)['"];?/g,
    (_, defaultName, moduleName) => {
      return `const ${defaultName} = ${moduleVar(moduleName)}.default ?? ${moduleVar(moduleName)};`;
    }
  );

  // import * as name from "vue"
  code = code.replace(
    /import\s+\*\s+as\s+([a-zA-Z0-9_$]+)\s+from\s+['"]([^'"]+)['"];?/g,
    (_, namespace, moduleName) => {
      return `const ${namespace} = ${moduleVar(moduleName)};`;
    }
  );

  // import defaultExport, { a, b as c } from "vue"
  code = code.replace(
    /import\s+([a-zA-Z0-9_$]+)\s*,\s*\{([^}]+)\}\s+from\s+['"]([^'"]+)['"];?/g,
    (_, defaultName, imports, moduleName) => {
      const names = imports
        .split(",")
        .map(s => s.trim())
        .filter(Boolean)
        .map(s => {
          if (s.includes(" as ")) {
            const [orig, alias] = s.split(" as ").map(x => x.trim());
            return `${orig}: ${alias}`;
          }
          return s;
        })
        .join(", ");
      return `const ${defaultName} = ${moduleVar(moduleName)}.default ?? ${moduleVar(moduleName)}; const { ${names} } = ${moduleVar(moduleName)};`;
    }
  );

  return code;
}

async function handleVueModule(getContentData: () => Promise<string>, path : string, options: Options) {
  // Leggi il contenuto del .vue
  const source = await getContentData();

  // Parsing del .vue
  const descriptor = parse(source, { filename: path }).descriptor;
  const id = btoa(path);
  
  // Compila script
  const script = compileScript(descriptor, { id });

  // Compila template
  let renderCode = '';
  if (descriptor.template) {
    const template = compileTemplate({
      source: descriptor.template.content,
      filename: path,
      id,
      compilerOptions: { mode: 'module', bindingMetadata: script?.bindings },
    });
    renderCode = template.code; // contiene export function render(...) {...}
  }

  // Compila stili
  for (const style of descriptor.styles) {
    const css = await compileStyleAsync({ source: style.content, filename: path, id });
    if (css.code) {
      const styleEl = document.createElement('style');
      styleEl.textContent = css.code;
      document.head.appendChild(styleEl);
    }
  }

  let scriptContent = script.content.replace(/export\s+default\s+/, "const __script = ");
  scriptContent = rewriteImportsToModuleCache(scriptContent);

  console.log(scriptContent)

  // Costruisci il modulo ES con iniezione dipendenze
  const finalModuleCode = `
    ${scriptContent}
    ${renderCode}
    __script.render = render;
    export default __script;
  `;

  // Trasforma con Babel Standalone
  const transformedCode = Babel.transform(finalModuleCode, {
    presets: [
      ['es2015', { modules: false }], // modules: false → lascia export/import
      'typescript'
    ],
    plugins: ['proposal-class-properties', "transform-typescript"],
    filename: 'file.js',
    sourceMaps: false
  }).code;

  console.log(transformedCode);

  const blob = new Blob([transformedCode], { type: 'application/javascript' });
  const url = URL.createObjectURL(blob);

  try {
    const mod = await import(url);
    return mod.default; // componente Vue pronto
  } finally {
    URL.revokeObjectURL(url);
  }
}

async function handleVueModuleCJS(
  getContentData: () => Promise<string>,
  path: string,
  moduleCache: Record<string, any>
) {
  const source = await getContentData();

  const descriptor = parse(source, { filename: path }).descriptor;
  const id = btoa(path);

  // Compila script
  const script = compileScript(descriptor, { id });

  // Compila template
  let renderCode = '';
  if (descriptor.template) {
    const template = compileTemplate({
      source: descriptor.template.content,
      filename: path,
      id,
      compilerOptions: { mode: 'module', bindingMetadata: script?.bindings },
    });
    renderCode = template.code.replace(/export function render/, 'function render'); 
    // render diventa funzione locale, poi la attacchiamo a module.exports
  }

  // Compila stili
  for (const style of descriptor.styles) {
    const css = await compileStyleAsync({ source: style.content, filename: path, id });
    if (css.code) {
      const styleEl = document.createElement('style');
      styleEl.textContent = css.code;
      document.head.appendChild(styleEl);
    }
  }

  // Riscrive gli import verso moduleCache
  let scriptContent = script.content.replace(/export\s+default\s+/, 'const __script = ');
  scriptContent = rewriteImportsToModuleCache(scriptContent, moduleCache);

  // Costruisce il CJS module
  const cjsCode = `
    ${scriptContent}
    ${renderCode}
    if (typeof render !== 'undefined') __script.render = render;
    module.exports = { default: __script, render: __script.render };
  `;

  const mod = { exports: {} };
  const fn = new Function('module', 'exports', 'moduleCache', cjsCode);
  fn(mod, mod.exports, moduleCache);

  return mod.exports.default;
}
